document.addEventListener('DOMContentLoaded', async () => { await filterProducts(); }); async function filterProducts() { try { const search = document.getElementById('search-input').value.toLowerCase(); const category = document.getElementById('cat-select').value; const municipio = document.getElementById('municipio-select').value; const products = await db.get('products'); const filtered = products.filter(p => { const matchSearch = p.name.toLowerCase().includes(search) || p.description.toLowerCase().includes(search); const matchCategory = !category || p.category === category; const matchMunicipio = !municipio || p.municipio === municipio; return matchSearch && matchCategory && matchMunicipio; }); renderProducts(filtered); } catch (error) { console.error('Error cargando productos:', error); document.getElementById('product-grid').innerHTML = '

Error al cargar productos. Por favor recarga la página.

'; } } function renderProducts(products) { const container = document.getElementById('product-grid'); if (products.length === 0) { container.innerHTML = '

No se encontraron productos.

'; return; } container.innerHTML = products.map(p => `

${p.name}

${ui.formatCurrency(p.price)}

${p.description}

📍 ${p.barrio}, ${p.municipio}
⭐ ${p.rating} (${p.stock} disponibles)

`).join(''); } async function addToCart(productId) { try { const product = await db.findOne('products', 'id', productId); if (product) { cart.add(product); } } catch (error) { console.error('Error añadiendo al carrito:', error); ui.toast('Error al añadir producto', 'error'); } }
